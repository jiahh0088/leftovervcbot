def init_db():
    """Initializes the SQLite database tables and structural schemas safely."""
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    
    # Individual User Standings Table
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS user_activity (
            guild_id TEXT,
            user_id TEXT,
            points INTEGER DEFAULT 0,
            voice_seconds INTEGER DEFAULT 0,
            PRIMARY KEY (guild_id, user_id)
        )
    """)
    
    cursor.execute("PRAGMA table_info(user_activity);")
    columns = [column[1] for column in cursor.fetchall()]
    if "voice_seconds" not in columns:
        cursor.execute("ALTER TABLE user_activity ADD COLUMN voice_seconds INTEGER DEFAULT 0;")
        
    conn.commit()
    conn.close()

def load_json(filename):
    if os.path.exists(filename):
        with open(filename, "r") as f:
            try: 
                return json.load(f)
            except json.JSONDecodeError: 
                return {}
    return {}

def save_json(filename, data):
    with open(filename, "w") as f:
        json.dump(data, f, indent=4)

def get_prefix(bot_instance, message):
    if not message.guild:
        return "!"
    config = load_json(CONFIG_FILE)
    return config.get(f"{message.guild.id}_prefix", "!")

def _apply_perm_bit(current_perms, **kwargs):
    perms = current_perms
    for perm, value in kwargs.items():
        setattr(perms, perm, value)
    return perms

def add_points(guild, member, points):
    guild_id = str(guild.id)
    user_id = str(member.id)
    
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    
    cursor.execute("""
        INSERT INTO user_activity (guild_id, user_id, points)
        VALUES (?, ?, ?)
        ON CONFLICT(guild_id, user_id) 
        DO UPDATE SET points = points + EXCLUDED.points
    """, (guild_id, user_id, points))
                
    conn.commit()
    conn.close()

def add_voice_time(guild, member, seconds):
    guild_id = str(guild.id)
    user_id = str(member.id)
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    
    cursor.execute("""
        INSERT INTO user_activity (guild_id, user_id, voice_seconds)
        VALUES (?, ?, ?, 0)
        ON CONFLICT(guild_id, user_id) 
        DO UPDATE SET voice_seconds = voice_seconds + EXCLUDED.voice_seconds
    """, (guild_id, user_id, seconds))
    
    conn.commit()
    conn.close()

def get_top_users(guild_id, limit=10):
    """Sorts entirely by unified points: Chat Points + (Voice Seconds / 30)."""
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    cursor.execute("""
        SELECT user_id, points, voice_seconds FROM user_activity 
        WHERE guild_id = ? 
        ORDER BY (points + (voice_seconds / 30)) DESC LIMIT ?
    """, (str(guild_id), limit))
    data = cursor.fetchall()
    conn.close()
    return data

def get_top_groups(guild, limit=5):
    """Dynamically aggregates unified activity points from members currently in specified roles."""
    config = load_json(CONFIG_FILE)
    guild_id = str(guild.id)
    fg_keys = [f"{guild_id}_fg_role1", f"{guild_id}_fg_role2", f"{guild_id}_fg_role3"]
    role_ids = [config.get(k) for k in fg_keys if config.get(k)]
    
    if not role_ids:
        return []
        
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    cursor.execute("SELECT user_id, points, voice_seconds FROM user_activity WHERE guild_id = ?", (guild_id,))
    users = cursor.fetchall()
    conn.close()
    
    group_scores = {int(rid): 0 for rid in role_ids}
    for u_id, chat_pts, v_secs in users:
        member = guild.get_member(int(u_id))
        if member:
            total_pts = chat_pts + (v_secs // 30)
            for r_id in role_ids:
                role = guild.get_role(int(r_id))
                if role and role in member.roles:
                    group_scores[int(r_id)] += total_pts
                    
    sorted_groups = sorted(group_scores.items(), key=lambda x: x[1], reverse=True)
    return sorted_groups[:limit]

def is_eligible_for_vc_tracking(state):
    return (
        state.channel is not None and 
        not state.self_mute and 
        not state.self_deaf and 
        not state.mute and 
        not state.deaf
    )

@bot.event
async def on_voice_state_update(member, before, after):
    if member.bot: return

    guild = member.guild
    guild_id = str(guild.id)
    user_id = str(member.id)
    config = load_json(CONFIG_FILE)

    was_eligible = is_eligible_for_vc_tracking(before)
    is_eligible = is_eligible_for_vc_tracking(after)

    if was_eligible and (not is_eligible or before.channel.id != after.channel.id):
        if user_id in voice_start_times:
            duration = discord.utils.utcnow() - voice_start_times[user_id]
            seconds = int(duration.total_seconds())
            if seconds > 0:
                add_voice_time(guild, member, seconds)
            del voice_start_times[user_id]

    if is_eligible and (not was_eligible or before.channel.id != after.channel.id):
        voice_start_times[user_id] = discord.utils.utcnow()

    if after.channel:
        portal_channel_id = config.get(f"{guild_id}_portal")
        if portal_channel_id and after.channel.id == portal_channel_id:
            destination_vcs = [vc for vc in guild.voice_channels if vc.id != portal_channel_id and vc.permissions_for(member).connect]
            if destination_vcs:
                try: 
                    await member.move_to(random.choice(destination_vcs))
                except discord.Forbidden:
                    try: 
                        await member.send("The Departure tried to take you, but administrative powers bound you to the portal.")
                    except discord.Forbidden: pass
            else:
                try: 
                    await member.move_to(None)
                except discord.Forbidden: pass
                try:
                    await member.send(embed=discord.Embed(title="Left Behind", description=f"You entered the portal in {guild.name}, but there were no open channels left.", color=PREMIUM_PINK))
                except discord.Forbidden: pass

def generate_user_embed(guild):
    embed = discord.Embed(title=f"{guild.name.upper()} TOP USERS", color=PREMIUM_PINK)
    top_users = get_top_users(guild.id, limit=10)
    
    if not top_users:
        embed.description = "No activity recorded yet."
        return embed

    lb_text = ""
    for i, (u_id, score, voice_seconds) in enumerate(top_users, 1):
        total_points = score + (voice_seconds // 30)
        lb_text += f"Rank {i:02d} | <@{u_id}>\nActivity Points: {total_points}\n\n"
        
    embed.description = lb_text
    return embed

def generate_fg_embed(guild):
    embed = discord.Embed(title=f"{guild.name.upper()} FG LEADERBOARD", color=PREMIUM_PINK)
    top_groups = get_top_groups(guild, limit=5)
    
    if not top_groups:
        embed.description = "No group logs allocated."
        return embed

    lb_text = ""
    for index, (r_id, score) in enumerate(top_groups, 1):
        role = guild.get_role(int(r_id))
        role_name = f"<@&{r_id}>" if role else "Null Allocation"
        lb_text += f"Group {index:02d} | {role_name}\nActivity Points: {score}\n\n"
        
    embed.description = lb_text
    return embed

@tasks.loop(seconds=30)
async def fast_update_loop():
    config = load_json(CONFIG_FILE)
    config_changed = False
    
    for guild in bot.guilds:
        guild_id = str(guild.id)
        await asyncio.sleep(0.5) 
        
        # 1. Update Persistent Individual Leaderboard Embed
        u_channel_id = config.get(f"{guild_id}_lb_channel")
        if u_channel_id:
            u_channel = guild.get_channel(int(u_channel_id))
            if u_channel:
                embed = generate_user_embed(guild)
                msg_id = config.get(f"{guild_id}_lb_msg_id")
                msg_verified = False
                
                if msg_id:
                    try:
                        msg = await u_channel.fetch_message(int(msg_id))
                        await msg.edit(embed=embed)
                        msg_verified = True
                    except (discord.NotFound, discord.HTTPException):
                        pass
                
                if not msg_verified:
                    try:
                        new_msg = await u_channel.send(embed=embed)
                        config[f"{guild_id}_lb_msg_id"] = new_msg.id
                        config_changed = True
                    except discord.HTTPException:
                        pass

        # 2. Update Persistent Group Leaderboard Embed
        g_channel_id = config.get(f"{guild_id}_fg_lb_channel")
        if g_channel_id:
            g_channel = guild.get_channel(int(g_channel_id))
            if g_channel:
                embed = generate_fg_embed(guild)
                msg_id = config.get(f"{guild_id}_fg_msg_id")
                msg_verified = False
                
                if msg_id:
                    try:
                        msg = await g_channel.fetch_message(int(msg_id))
                        await msg.edit(embed=embed)
                        msg_verified = True
                    except (discord.NotFound, discord.HTTPException):
                        pass
                
                if not msg_verified:
                    try:
                        new_msg = await g_channel.send(embed=embed)
                        config[f"{guild_id}_fg_msg_id"] = new_msg.id
                        config_changed = True
                    except discord.HTTPException:
                        pass

        # 3. Synchronize Standard Top-3 Active Placements 
        r1_id, r2_id, r3_id = config.get(f"{guild_id}_role1"), config.get(f"{guild_id}_role2"), config.get(f"{guild_id}_role3")
        roles = [guild.get_role(int(r1_id)) if r1_id else None, 
                 guild.get_role(int(r2_id)) if r2_id else None, 
                 guild.get_role(int(r3_id)) if r3_id else None]
        
        top_three = get_top_users(guild_id, limit=3)
        if top_three:
            for r in roles:
                if r:
                    for m in r.members:
                        if m.id not in [int(uid) for uid, _, _ in top_three]:
                            try: await m.remove_roles(r)
                            except: pass
            
            for rank, (u_id, _, _) in enumerate(top_three):
                if rank < len(roles) and roles[rank]:
                    try:
                        member = guild.get_member(int(u_id)) or await guild.fetch_member(int(u_id))
                        if member and roles[rank] not in member.roles:
                            await member.add_roles(roles[rank])
                    except: pass

        # 4. Synchronize VC Milestone Accumulation Roles (1h, 5h, 25h, 35h, 55h)
        vc_tier_ids = [
            config.get(f"{guild_id}_vc_tier1"),      # 1h
            config.get(f"{guild_id}_vc_tier2"),      # 5h
            config.get(f"{guild_id}_vc_protected"),  # 25h
            config.get(f"{guild_id}_vc_tierx"),      # 35h
            config.get(f"{guild_id}_vc_vip")         # 55h
        ]
        vc_milestone_roles = [guild.get_role(int(rid)) if rid else None for rid in vc_tier_ids]
        thresholds = [3600, 18000, 90000, 126000, 198000]

        conn = sqlite3.connect(DB_FILE)
        cursor = conn.cursor()
        cursor.execute("SELECT user_id, voice_seconds FROM user_activity WHERE guild_id = ?", (guild_id,))
        guild_records = cursor.fetchall()
        conn.close()

        for u_id, total_seconds in guild_records:
            try:
                member = guild.get_member(int(u_id)) or await guild.fetch_member(int(u_id))
                if not member: continue
                
                for idx, milestone_role in enumerate(vc_milestone_roles):
                    if milestone_role:
                        if total_seconds >= thresholds[idx]:
                            if milestone_role not in member.roles:
                                await member.add_roles(milestone_role)
                        else:
                            if milestone_role in member.roles:
                                await member.remove_roles(milestone_role)
            except:
                pass

    if config_changed:
        save_json(CONFIG_FILE, config)
