async def init_db():
    """Initializes activity tracking, group mappings, confessions, and economics with correct schemas."""
    async with aiosqlite.connect(DB_FILE) as db:
        # Core User Standings, Economy Vaults, and Cooldown Ledger
        await db.execute("""
            CREATE TABLE IF NOT EXISTS user_activity (
                guild_id TEXT,
                user_id TEXT,
                points INTEGER DEFAULT 0,
                bank INTEGER DEFAULT 0,
                daily_norm INTEGER DEFAULT 10,
                daily_progress INTEGER DEFAULT 0,
                checked_in INTEGER DEFAULT 0,
                last_fg_switch TEXT DEFAULT NULL,
                PRIMARY KEY (guild_id, user_id)
            )
        """)
        
        # Group Activity Accumulator Mapping
        await db.execute("""
            CREATE TABLE IF NOT EXISTS group_activity (
                guild_id TEXT,
                role_id TEXT,
                points INTEGER DEFAULT 0,
                PRIMARY KEY (guild_id, role_id)
            )
        """)
        
        # Anonymous Confessions Mapping Tracker
        await db.execute("""
            CREATE TABLE IF NOT EXISTS confessions (
                guild_id TEXT,
                confession_id INTEGER PRIMARY KEY AUTOINCREMENT,
                author_id TEXT,
                content TEXT,
                timestamp TEXT
            )
        """)
        await db.commit()

async def load_json(filename):
    async with json_lock:
        if os.path.exists(filename):
            with open(filename, "r") as f:
                try: 
                    return json.load(f)
                except json.JSONDecodeError: 
                    return {}
        return {}

async def save_json(filename, data):
    async with json_lock:
        with open(filename, "w") as f:
            json.dump(data, f, indent=4)

async def get_prefix(bot_instance, message):
    if not message.guild: 
        return "!"
    config = await load_json(CONFIG_FILE)
    return config.get(f"{message.guild.id}_prefix", "!")

        await ctx.send("⚠️ You are missing required parameters for this command.")
    elif isinstance(error, commands.BadArgument):
        await ctx.send("⚠️ Invalid argument provided. (e.g., using letters instead of numbers)")

        await ctx.send(f"⏳ System cooling down. Try again in {int(error.retry_after)}s.")

        print(f"Unhandled Execution Error: {error}")

async def get_user_data(guild_id, user_id):
    async with aiosqlite.connect(DB_FILE) as db:
        async with db.execute("SELECT points, bank, daily_norm, daily_progress, checked_in, last_fg_switch FROM user_activity WHERE guild_id = ? AND user_id = ?", (str(guild_id), str(user_id))) as cursor:
            row = await cursor.fetchone()
        if not row:
            await db.execute("INSERT INTO user_activity (guild_id, user_id) VALUES (?, ?)", (str(guild_id), str(user_id)))
            await db.commit()
            row = (0, 0, 10, 0, 0, None)
    return {"points": row[0], "bank": row[1], "daily_norm": row[2], "daily_progress": row[3], "checked_in": row[4], "last_fg_switch": row[5]}

async def add_points(guild, member, amount):
    guild_id = str(guild.id)
    user_id = str(member.id)
    
    data = await get_user_data(guild_id, user_id)
    new_pts = data["points"] + amount
    new_prog = data["daily_progress"] + amount
    
    async with aiosqlite.connect(DB_FILE) as db:
        await db.execute("""
            UPDATE user_activity 
            SET points = ?, daily_progress = ?
            WHERE guild_id = ? AND user_id = ?
        """, (new_pts, new_prog, guild_id, user_id))
        
        if not data["checked_in"] and new_prog >= math.ceil(data["daily_norm"] * 0.25):
            bonus = math.ceil(new_pts * 0.05)
            new_pts += bonus
            await db.execute("""
                UPDATE user_activity 
                SET points = ?, checked_in = 1
                WHERE guild_id = ? AND user_id = ?
            """, (new_pts, guild_id, user_id))
            
        # Sync matching Friend Group roles data mappings dynamically
        config = await load_json(CONFIG_FILE)
        roles_list = config.get(f"{guild_id}_fg_roles_list", [])
        for rid in roles_list:
            role = guild.get_role(int(rid))
            if role and role in member.roles:
                await db.execute("""
                    INSERT INTO group_activity (guild_id, role_id, points)
                    VALUES (?, ?, ?)
                    ON CONFLICT(guild_id, role_id) 
                    DO UPDATE SET points = points + EXCLUDED.points
                """, (guild_id, str(rid), amount))
                
        await db.commit()

async def modify_bank(guild_id, user_id, amount):

        await db.commit()

#      DYNAMIC ROLE REPLACEMENT ROUTINE
# ==========================================

async def replace_fg_role(member: discord.Member, new_role_id: int, roles_list: list) -> bool:
    """Removes existing tracking group roles from user before assigning the target."""
    to_remove = [member.guild.get_role(int(rid)) for rid in roles_list if int(rid) != new_role_id and member.guild.get_role(int(rid)) in member.roles]
    target_role = member.guild.get_role(new_role_id)
    
    if not target_role: 
        return False
    try:
        if to_remove:
            await member.remove_roles(*[r for r in to_remove if r])
        await member.add_roles(target_role)
        return True
    except discord.Forbidden:
        return False

async def purchase_processor(ctx, item_type: str, target: discord.Member = None, *, aux_args: str = None):
    if not target: 
        return await ctx.send("Specify a recipient handle.")
    if target.bot:
        return await ctx.send("Automated targets immune to privilege execution paths.")

    total_wealth = user_data["points"] + user_data["bank"]

        cost = max(50000, math.ceil(total_wealth * 0.10))
        if user_data["points"] < cost: 
            return await ctx.send(f"Liquid assets insufficient. Cost evaluated to: `{cost}`")
        try:
            await target.timeout(timedelta(minutes=5), reason="Marketplace Privilege Purchase Execution.")

        except discord.Forbidden: 
            await ctx.send("Permissions hierarchy levels block command execution vectors.")
        
    elif item_type.lower() in ["nickname", "nick"]:
        if not aux_args: 
            return await ctx.send("Provide the new name string argument parameters.")
        if target.guild_permissions.administrator: 
            cost = max(35000, math.ceil(total_wealth * 0.10))
        elif target.guild_permissions.manage_messages: 
            cost = 25000
        else: 
            cost = 10000
            
        if user_data["points"] < cost: 
            return await ctx.send(f"Asset parameters shortfall. Requirement: `{cost}` points.")
        try:
            await target.edit(nick=aux_args[:32])
            await add_points(ctx.guild, ctx.author, -cost)
            await ctx.send(f"Identity changes applied to {target.mention}. Balance deducted: `{cost}` tokens.")
        except discord.Forbidden: 
            await ctx.send("Security configurations prohibit renaming this target profile.")

class MinesGameView(discord.ui.View):
    def __init__(self, ctx, bet, bombs_count):
        super().__init__(timeout=60)
        self.ctx = ctx
        self.bet = bet
        self.bombs_count = bombs_count
        self.grid = ["safe"] * 25
        self.revealed = [False] * 25
        self.clicks = 0
        deployed = 0
        while deployed < bombs_count:
            idx = random.randint(0, 24)
            if self.grid[idx] == "safe":
                self.grid[idx] = "bomb"
                deployed += 1
        for i in range(25): 
            self.add_item(MinesButton(i, row=i // 5))

    def calculate_multiplier(self):
        k = self.clicks
        if k == 0: 
            return 1.0
        num = math.comb(25 - self.bombs_count, k)
        den = math.comb(25, k)
        if den == 0 or num == 0: 
            return 20.0
        return round((1.0 / (num / den)) * 0.95, 2)

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        if interaction.user.id != self.ctx.author.id:
            await interaction.response.send_message("This isn't your active wager layout.", ephemeral=True)

class BlackjackView(discord.ui.View):
    def __init__(self, ctx, bet):
        super().__init__(timeout=60)
        self.ctx = ctx
        self.bet = bet
        self.deck = [2,3,4,5,6,7,8,9,10,10,10,10,11] * 4
        random.shuffle(self.deck)
        self.player_hand = [self.draw(), self.draw()]
        self.dealer_hand = [self.draw(), self.draw()]

    def draw(self): 
        return self.deck.pop()
        
    def eval_hand(self, hand):
        score = sum(hand)
        while score > 21 and 11 in hand: 
            hand[hand.index(11)] = 1
            score = sum(hand)
        return score
        
    def render_embed(self, finalized=False):
        emb = discord.Embed(title="Blackjack Operational Suite", color=discord.Color.gold())
        emb.add_field(name="Player Score", value=f"Cards: {self.player_hand}\nEvaluation metric: `{self.eval_hand(self.player_hand)}`", inline=True)
        if finalized: 
            emb.add_field(name="Dealer Score", value=f"Cards: {self.dealer_hand}\nEvaluation metric: `{self.eval_hand(self.dealer_hand)}`", inline=True)
        else: 
            emb.add_field(name="Dealer Score", value=f"Cards: [{self.dealer_hand[0]}, ?]", inline=True)
        return emb
        
    async def interaction_check(self, interaction: discord.Interaction) -> bool: 
        if interaction.user.id != self.ctx.author.id:
            await interaction.response.send_message("This isn't your card table.", ephemeral=True)
            return False
        return True

    @discord.ui.button(label="Hit", style=discord.ButtonStyle.blurple)
    async def hit_node(self, interaction: discord.Interaction, button: discord.ui.Button):
        self.player_hand.append(self.draw())
        if self.eval_hand(self.player_hand) > 21:
            for item in self.children: 
                item.disabled = True
            self.stop()
            # Upfront deduction already occurred.
            await interaction.response.edit_message(content="💥 Hand evaluation threshold busted! Wager lost.", embed=self.render_embed(True), view=self)
        else: 
            await interaction.response.edit_message(embed=self.render_embed(), view=self)

    @discord.ui.button(label="Stand", style=discord.ButtonStyle.grey)
    async def stand_node(self, interaction: discord.Interaction, button: discord.ui.Button):
        for item in self.children: 
            item.disabled = True
        while self.eval_hand(self.dealer_hand) < 17: 
            self.dealer_hand.append(self.draw())
        
        p_total, d_total = self.eval_hand(self.player_hand), self.eval_hand(self.dealer_hand)
        if d_total > 21 or p_total > d_total: 
            payout = self.bet * 2 # Refund + winnings
            await add_points(interaction.guild, self.ctx.author, payout)
            msg = f"🏆 Superiority established! Dispatched: `{payout}` total points."
        elif p_total < d_total: 
            msg = f"📉 Deficit incurred. Lost wager."
        else: 
            await add_points(interaction.guild, self.ctx.author, self.bet) # Refund
            msg = "⚖️ Evaluation data push. Wagers pushed back balanced."
        self.stop()
        await interaction.response.edit_message(content=msg, embed=self.render_embed(True), view=self)

# --- TIC TAC TOE CLAN COMBAT ARENA ---
class TicTacToeView(discord.ui.View):
    def __init__(self, ctx, challenger, target_role, challenge_role, bet_points):
        super().__init__(timeout=90)
        self.ctx = ctx
        self.challenger = challenger
        self.target_role = target_role
        self.challenge_role = challenge_role
        self.bet_points = bet_points
        self.grid = [None] * 9
        self.current_turn = challenger
        self.defender = None
        for i in range(9): 
            self.add_item(TicTacToeButton(i))
            
    async def check_win(self):
        lines = [(0,1,2),(3,4,5),(6,7,8),(0,3,6),(1,4,7),(2,5,8),(0,4,8),(2,4,6)]
        for r in lines:
            if self.grid[r[0]] and self.grid[r[0]] == self.grid[r[1]] == self.grid[r[2]]: 
                return self.grid[r[0]]
        if None not in self.grid: 
            return "Tie"
        return None

class TicTacToeButton(discord.ui.Button):
    def __init__(self, index): 
        super().__init__(label="-", style=discord.ButtonStyle.grey, row=index // 3)
        self.index = index
        
    async def callback(self, interaction: discord.Interaction):
        view: TicTacToeView = self.view
        
        # Defender claim phase
        if not view.defender:
            if view.target_role not in [r.id for r in interaction.user.roles]: 
                return await interaction.response.send_message("Unauthorized interaction layer vector.", ephemeral=True)
            
            # Ensure defender has enough points to match the wager
            def_data = await get_user_data(interaction.guild.id, interaction.user.id)
            if def_data["points"] < view.bet_points:
                return await interaction.response.send_message("You lack the point balance to accept this wager.", ephemeral=True)
                
            view.defender = interaction.user
            # Deduct defender bet upfront
            await add_points(interaction.guild, interaction.user, -view.bet_points)
            
        if interaction.user.id != view.current_turn.id: 
            return await interaction.response.send_message("Awaiting opposing coordination matrix validations data paths.", ephemeral=True)
            
        mark = "X" if interaction.user.id == view.challenger.id else "O"
        view.grid[self.index] = mark
        self.label = mark
        self.disabled = True
        self.style = discord.ButtonStyle.blurple if mark == "X" else discord.ButtonStyle.green
        
        outcome = await view.check_win()
        if outcome:
            for item in view.children: 
                item.disabled = True
            view.stop()
            
            if outcome == "Tie": 
                # Refund both users
                await add_points(interaction.guild, view.challenger, view.bet_points)
                await add_points(interaction.guild, view.defender, view.bet_points)
                await interaction.response.edit_message(content="⚖️ Stalemate coordinate arrays stabilized. Wager balances held even and refunded.", view=view)
            else:
                winner = view.challenger if outcome == "X" else view.defender
                w_fg = view.challenge_role if outcome == "X" else view.target_role
                
                # Give winner the pot (their bet back + loser's bet)
                await add_points(interaction.guild, winner, view.bet_points * 2)
                await interaction.response.edit_message(content=f"👑 **VICTORY SECURED!** {winner.mention} wins the board. Added total pot of `{view.bet_points * 2}` points over to <@&{w_fg}>.", view=view)
        else: 
            view.current_turn = view.defender if view.current_turn.id == view.challenger.id else view.challenger
            await interaction.response.edit_message(content=f"🎮 Arena grid active. Wagers locked. Awaiting turn deployment vector from: {view.current_turn.mention}", view=view)

    prefix = await get_prefix(bot, message)
    if message.content.startswith(prefix): 
        await bot.process_commands(message)
    else:
        config = await load_json(CONFIG_FILE)
        welcome_role_id = config.get(f"{message.guild.id}_welcome_role")
        if welcome_role_id and welcome_role_id in [r.id for r in message.author.roles]: 
            await add_points(message.guild, message.author, 2)
        else: 
            await add_points(message.guild, message.author, 1)

@bot.event
async def on_voice_state_update(member, before, after):
    if member.bot: 
        return
    guild, guild_id, user_id = member.guild, str(member.guild.id), str(member.id)
    config = await load_json(CONFIG_FILE)
    session_key = user_id

    if before.channel and (not after.channel or before.channel.id != after.channel.id):
        if session_key in voice_start_times:
            duration = discord.utils.utcnow() - voice_start_times[session_key]
            minutes = int(duration.total_seconds() / 60)
            if minutes > 0:
                base_pts = minutes * 2
                roles_list = config.get(f"{guild_id}_fg_roles_list", [])
                matching_fg_id = None
                for r in member.roles:
                    if r.id in roles_list: 
                        matching_fg_id = r.id
                        break
                if matching_fg_id:
                    if sum(1 for m in before.channel.members if matching_fg_id in [r.id for r in m.roles]) >= 8: 
                        base_pts *= 4
                await add_points(guild, member, base_pts)
            voice_start_times.pop(session_key, None)

    if after.channel and (not before.channel or before.channel.id != after.channel.id):
        voice_start_times[session_key] = discord.utils.utcnow()
        portal_id = config.get(f"{guild_id}_portal")
        if portal_id and after.channel.id == portal_id:
            targets = [vc for vc in guild.voice_channels if vc.id != portal_id and vc.permissions_for(member).connect]
            if targets:
                try: 
                    await member.move_to(random.choice(targets))
                except discord.Forbidden: 
                    pass
            else:
                try: 
                    await member.move_to(None)
                except discord.Forbidden: 
                    pass

@tasks.loop(seconds=30)
async def fast_update_loop():
    config = await load_json(CONFIG_FILE)
    config_changed = False
    
    for guild in bot.guilds:
        guild_id = str(guild.id)
        await asyncio.sleep(0.1)  # Prevent blocking Event Loop during guild iteration
        
        # --- 1. DYNAMIC PARENT CATEGORY VIP ROOM SETUPS ---
        try:
            top_groups_data = await get_top_groups(guild_id, limit=1)
            category_destination_id = config.get(f"{guild_id}_vip_category_id")
            category_obj = guild.get_channel(category_destination_id) if category_destination_id else None
            
            if top_groups_data:
                winner_fg_id = int(top_groups_data[0][0])
                vip_channel_id = config.get(f"{guild_id}_vip_vc_id")
                vip_channel = guild.get_channel(vip_channel_id) if vip_channel_id else None
                
                if not vip_channel:
                    try:
                        overwrites = {
                            guild.default_role: discord.PermissionOverwrite(connect=False),
                            guild.get_role(winner_fg_id): discord.PermissionOverwrite(connect=True, speak=True)
                        }
                        target_bitrate = min(384000, guild.bitrate_limit)
                        vip_channel = await guild.create_voice_channel(name="👑 VIP Lounge", category=category_obj, overwrites=overwrites, bitrate=target_bitrate)
                        config[f"{guild_id}_vip_vc_id"] = vip_channel.id
                        config_changed = True
                    except Exception: 
                        pass
                else:
                    try:
                        role_obj = guild.get_role(winner_fg_id)
                        if role_obj:
                            overwrites = {
                                guild.default_role: discord.PermissionOverwrite(connect=False),
                                role_obj: discord.PermissionOverwrite(connect=True, speak=True)
                            }
                            await vip_channel.edit(category=category_obj, overwrites=overwrites)
                    except Exception: 
                        pass
        except Exception as e:
            print(f"VIP Setup Segment Failure: {e}")

        # --- 2. DYNAMIC CATEGORY OVERRIDES FOR RANK 1 MEMBER ---
        top_three = await get_top_users(guild_id, limit=3)
        try:
            if top_three and category_obj:
                rank_one_user_id = int(top_three[0][0])
                rank_one_member = guild.get_member(rank_one_user_id)
                if rank_one_member:
                    cat_overwrites = category_obj.overwrites
                    cat_overwrites[rank_one_member] = discord.PermissionOverwrite(mute_members=True, deafen_members=True, move_members=True)
                    await category_obj.edit(overwrites=cat_overwrites)
        except Exception as e:
            print(f"Category Override Segment Failure: {e}")

        # --- 3. RANDOM MEET-UP PING V2 ---
        try:
            server_is_completely_dark = True
            for voice_channel in guild.voice_channels:
                if len(voice_channel.members) > 0:
                    server_is_completely_dark = False
                    break 
            
            if not server_is_completely_dark:
                last_voice_activity[guild_id] = datetime.utcnow()
            else:
                last_act = last_voice_activity.get(guild_id, datetime.utcnow())
                if (datetime.utcnow() - last_act).total_seconds() > 14400: 
                    fg_ranks = await get_top_groups(guild_id, limit=3)
                    if fg_ranks:
                        selected_fg_role_id = int(random.choice(fg_ranks)[0])
                        ping_channel_id = config.get(f"{guild_id}_lb_channel")
                        ping_channel = guild.get_channel(ping_channel_id) if ping_channel_id else None
                        if ping_channel:
                            await ping_channel.send(f"🚨 **GLOBAL RALLY PING (V2).** All server voice channels have been dark for 4+ continuous hours. Group alliance <@&{selected_fg_role_id}> is ordered to mobilize lines instantly.")
                            last_voice_activity[guild_id] = datetime.utcnow()
        except Exception as e:
            print(f"Rally Ping Segment Failure: {e}")

        # --- 4. INDIVIDUAL LIVE LEADERBOARD INTERACTION PIPELINES ---
        try:
            u_chan_id = config.get(f"{guild_id}_lb_channel")
            if u_chan_id:
                u_chan = guild.get_channel(int(u_chan_id))
                if u_chan:
                    top_users = await get_top_users(guild_id, limit=10)
                    emb = discord.Embed(title=f"{guild.name.upper()} // INDIVIDUAL STANDINGS", color=discord.Color.black())
                    txt = ""
                    for i, (uid, sc) in enumerate(top_users, 1): 
                        txt += f"Rank {i:02d} | <@{uid}>\nScore: {sc} points\n\n"
                    emb.description = txt or "No ledger mappings records allocated."
                    
                    msg_id = config.get(f"{guild_id}_lb_msg_id")
                    msg_verified = False
                    if msg_id:
                        try: 
                            m = u_chan.get_partial_message(int(msg_id))
                            await m.edit(embed=emb)
                            msg_verified = True
                        except discord.HTTPException: 
                            pass
                    if not msg_verified:
                        try: 
                            nm = await u_chan.send(embed=emb)
                            config[f"{guild_id}_lb_msg_id"] = nm.id
                            config_changed = True
                        except discord.HTTPException: 
                            pass
        except Exception as e:
            print(f"User Leaderboard Segment Failure: {e}")

        # --- 5. GROUP LIVE LEADERBOARD INTERACTION PIPELINES ---
        try:
            g_chan_id = config.get(f"{guild_id}_fg_lb_channel")
            if g_chan_id:
                g_chan = guild.get_channel(int(g_chan_id))
                if g_chan:
                    fg_data = await get_top_groups(guild_id, limit=15)
                    emb = discord.Embed(title=f"{guild.name.upper()} // GROUP STANDINGS", color=discord.Color.black())
                    txt = ""
                    for i, (rid, sc) in enumerate(fg_data, 1): 
                        txt += f"Group {i:02d} | <@&{rid}>\nIndex: {sc} cumulative\n\n"
                    emb.description = txt or "No active group definitions metrics logged."
                    
                    msg_id = config.get(f"{guild_id}_fg_msg_id")
                    msg_verified = False
                    if msg_id:
                        try: 
                            m = g_chan.get_partial_message(int(msg_id))
                            await m.edit(embed=emb)
                            msg_verified = True
                        except discord.HTTPException: 
                            pass
                    if not msg_verified:
                        try: 
                            nm = await g_chan.send(embed=emb)
                            config[f"{guild_id}_fg_msg_id"] = nm.id
                            config_changed = True
                        except discord.HTTPException: 
                            pass
        except Exception as e:
            print(f"Group Leaderboard Segment Failure: {e}")

        # --- 6. ADVANCED PLACEMENTS PROGRESSIVE ROLE SYNCING ---
        try:
            r1_id, r2_id, r3_id = config.get(f"{guild_id}_role1"), config.get(f"{guild_id}_role2"), config.get(f"{guild_id}_role3")
            roles = [guild.get_role(int(r1_id)) if r1_id else None, guild.get_role(int(r2_id)) if r2_id else None, guild.get_role(int(r3_id)) if r3_id else None]
            if top_three:
                for r in roles:
                    if r:
                        for m in r.members:
                            if m.id not in [int(uid) for uid, _ in top_three]:
                                try: 
                                    await m.remove_roles(r)
                                except Exception: 
                                    pass
                for rank, (uid, _) in enumerate(top_three):
                    if rank < len(roles) and roles[rank]:
                        try:
                            member = guild.get_member(int(uid))
                            if member and roles[rank] not in member.roles: 
                                await member.add_roles(roles[rank])
                        except Exception: 
                            pass
        except Exception as e:
            print(f"Role Syncing Segment Failure: {e}")

    if config_changed: 
        await save_json(CONFIG_FILE, config)

@tasks.loop(hours=24)
async def daily_reset_loop():
    async with aiosqlite.connect(DB_FILE) as db:
        async with db.execute("SELECT guild_id, user_id, daily_progress, daily_norm FROM user_activity") as cursor:
            records = await cursor.fetchall()
            
        for gid, uid, dp, dn in records:
            await db.execute("UPDATE user_activity SET daily_norm = ?, daily_progress = 0, checked_in = 0 WHERE guild_id = ? AND user_id = ?", (max(10, math.ceil((dn + dp) / 2)), gid, uid))
        await db.commit()

@bot.command(name="triggerweeklyreset")
@commands.has_permissions(administrator=True)
async def execute_weekly_reset_sequence(ctx):
    config = await load_json(CONFIG_FILE)
    roles_list = config.get(f"{ctx.guild.id}_fg_roles_list", [])
    if not roles_list: 
        return await ctx.send("No registered friend groups structures found.")
    await ctx.send("⌛ Calculating weekly dividend metrics updates and resetting point matrix systems...")
    
    async with aiosqlite.connect(DB_FILE) as db:
        for r_id in roles_list:
            role = ctx.guild.get_role(int(r_id))
            if not role or not role.members: 
                continue
            
            total_points, member_ids = 0, [str(m.id) for m in role.members]
            for m_id in member_ids:
                async with db.execute("SELECT points FROM user_activity WHERE guild_id = ? AND user_id = ?", (str(ctx.guild.id), m_id)) as cursor:
                    row = await cursor.fetchone()
                if row: 
                    total_points += row[0]
                    
            dividend = math.ceil(total_points / len(member_ids))
            for m_id in member_ids:
                await db.execute("INSERT INTO user_activity (guild_id, user_id, bank) VALUES (?, ?, ?) ON CONFLICT(guild_id, user_id) DO UPDATE SET bank = bank + EXCLUDED.bank", (str(ctx.guild.id), m_id, dividend))
                
        await db.execute("UPDATE user_activity SET points = 0 WHERE guild_id = ?", (str(ctx.guild.id),))
        await db.execute("UPDATE group_activity SET points = 0 WHERE guild_id = ?", (str(ctx.guild.id),))
        await db.commit()
        
    await ctx.send("🏆 **WEEKLY ALLIANCE PAYOUT MATRIX RESET COMPLETE.** Dividend returns split cleanly across vaults.")
